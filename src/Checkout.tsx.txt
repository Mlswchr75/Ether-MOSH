import { useEffect } from "react";
import { Link, useNavigate, useSearchParams } from "react-router-dom";
import { Helmet } from "react-helmet-async";
import { ArrowLeft } from "lucide-react";
import { useAuth } from "@/hooks/useAuth";
import { PaymentTestModeBanner } from "@/components/PaymentTestModeBanner";
import { StripeEmbeddedCheckout } from "@/components/StripeEmbeddedCheckout";

const VALID_PRICES = new Set([
  "mosh_supporter_once",
  "mosh_tip_1",
  "mosh_tip_5",
  "mosh_tip_25",
]);

const Checkout = () => {
  const { user, loading } = useAuth();
  const [params] = useSearchParams();
  const navigate = useNavigate();
  const priceId = params.get("price") ?? "";

  useEffect(() => {
    if (loading) return;
    if (!user) {
      const next = encodeURIComponent(`/checkout?price=${priceId}`);
      navigate(`/auth?next=${next}`, { replace: true });
    }
  }, [user, loading, priceId, navigate]);

  const valid = VALID_PRICES.has(priceId);

  return (
    <main className="min-h-screen bg-background text-foreground">
      <Helmet>
        <title>Checkout — MOSH</title>
        <meta name="description" content="Complete your MOSH purchase securely with Stripe." />
        <meta name="robots" content="noindex" />
      </Helmet>

      <PaymentTestModeBanner />

      <header className="flex items-center justify-between px-5 pt-6">
        <Link
          to="/pricing"
          className="flex items-center gap-2 font-mono text-xs uppercase tracking-[0.25em] text-muted-foreground hover:text-accent transition"
        >
          <ArrowLeft className="h-3 w-3" /> back to pricing
        </Link>
      </header>

      <section className="mx-auto w-full max-w-3xl px-5 py-10">
        <h1 className="mb-6 font-sans text-3xl font-bold tracking-tight">Checkout</h1>

        {!valid ? (
          <div className="rounded-lg border border-border/60 p-6">
            <p className="text-sm text-foreground/75">
              We couldn&apos;t find that item.{" "}
              <Link to="/pricing" className="text-accent hover:underline">
                Return to pricing
              </Link>
              .
            </p>
          </div>
        ) : loading || !user ? (
          <p className="font-mono text-[11px] uppercase tracking-[0.25em] text-muted-foreground">
            preparing checkout…
          </p>
        ) : (
          <StripeEmbeddedCheckout
            priceId={priceId}
            customerEmail={user.email ?? undefined}
            userId={user.id}
            returnUrl={`${window.location.origin}/pricing?checkout=success`}
          />
        )}
      </section>
    </main>
  );
};

export default Checkout;
